    75	/** @param {string} region @returns {{verdict: string, findings: string} | null} */
    76	function lastVerdictIn(region) {
    77	  let last = null;
    78	  for (const m of region.matchAll(VERDICT_RE)) last = m;
    79	  if (!last) return null;
    80	  const after = region.slice((last.index ?? 0) + last[0].length);

    84	/**
    85	 * Region-bounded verdict parsing — the prompt region (which quotes the
    86	 * verdict format in its instructions) is NEVER scanned, and there is no
    87	 * whole-transcript fallback. Unparseable is BLOCKED, never APPROVED.
    88	 * - codex: only the text after the LAST line starting with 'tokens used'.
    89	 * - claude-code: only the `result` field of the structured JSON output.
    90	 * Multiple tails inside the trusted region → the LAST one wins.
    91	 * @param {string} transcript @param {{platform: string}} opts
    92	 * @returns {{verdict: string, findings: string, reason?: string}}
    93	 */
    94	export function parseVerdict(transcript, { platform }) {
    95	  const unparseable = { verdict: 'BLOCKED', findings: '', reason: 'unparseable' };
    96	  const text = String(transcript ?? '');
    97	
    98	  if (platform === 'claude-code') {
    99	    let obj = null;
   100	    try {
   101	      obj = JSON.parse(text.trim());
   102	    } catch {
   103	      return unparseable;
   104	    }
   105	    if (typeof obj?.result !== 'string') return unparseable;
   106	    const hit = lastVerdictIn(obj.result);
   107	    return hit ?? unparseable;
   108	  }
   109	
   110	  // codex: locate the LAST 'tokens used' marker line; the trusted region is
   111	  // strictly after it.
   112	  const lines = text.split('\n');
   113	  let markerLine = -1;
   114	  for (let i = 0; i < lines.length; i += 1) {
   115	    if (/^tokens used\b/i.test(lines[i])) markerLine = i;
   116	  }
   117	  if (markerLine === -1) return unparseable;
   118	  const region = lines.slice(markerLine + 1).join('\n');
   119	  const hit = lastVerdictIn(region);
   120	  return hit ?? unparseable;
   121	}

   123	const TRUNCATION_MARKER = '\n[... truncated by baton log cap ...]\n';
   124	
   125	/**
   126	 * Cap a transcript: head preserved, middle replaced with a marker, TAIL
   127	 * preserved verbatim (the verdict lives at the end and must never be cut).
   128	 * @param {string} text @param {number} maxBytes
   129	 * @returns {string}
   130	 */
   131	export function capLog(text, maxBytes) {
   132	  const s = String(text ?? '');
   133	  if (Buffer.byteLength(s) <= maxBytes) return s;
   134	  const headBytes = Math.max(1, Math.floor(maxBytes * 0.4));
   135	  const tailBytes = Math.max(1, maxBytes - headBytes);
   136	  const buf = Buffer.from(s);
   137	  const head = buf.subarray(0, headBytes).toString();
   138	  const tail = buf.subarray(buf.length - tailBytes).toString();
   139	  return head + TRUNCATION_MARKER + tail;
   140	}

   152	/** @param {number} pid @param {NodeJS.Signals} sig */
   153	function killGroup(pid, sig) {
   154	  try {
   155	    process.kill(-pid, sig);
   156	  } catch {

   161	/**
   162	 * Spawn and supervise one child: detached in its OWN process group, stdin
   163	 * closed; on timeout SIGTERM the group, wait graceMs, SIGKILL the group (no
   164	 * zombie grandchildren). Output streams to a capped log at opts.logPath; the
   165	 * verdict is parsed from the full transcript, region-bounded.
   166	 * @param {{command: string, args: string[], env?: Record<string, string>, cwd?: string}} spec
   167	 * @param {{timeoutMs: number, graceMs: number, logPath: string, maxLogBytes?: number, platform?: string, onStart?: (info: {pid: number, pgid: number}) => void}} opts
   168	 * @returns {Promise<{timedOut: boolean, exitCode: number | null, verdict?: string, findings?: string, reason?: string, logPath: string, pgid?: number}>}
   169	 */
   170	export function superviseChild(spec, opts) {
   171	  return new Promise((resolve) => {
   172	    const maxLogBytes = opts.maxLogBytes ?? 1_000_000;
   173	    const child = spawn(spec.command, spec.args, {
   174	      cwd: spec.cwd,
   175	      env: { ...process.env, ...(spec.env ?? {}) },
   176	      stdio: ['ignore', 'pipe', 'pipe'],
   177	      detached: true, // its own process group — the kill unit
   178	    });
   179	
   180	    // Report the group at spawn time so the supervisor can persist it BEFORE
   181	    // the child completes — a supervisor crash leaves the orphan reap-able.
   182	    const pgid = typeof child.pid === 'number' ? child.pid : -1;
   183	    if (typeof opts.onStart === 'function' && typeof child.pid === 'number') {
   184	      opts.onStart({ pid: child.pid, pgid });
   185	    }
   186	
   187	    let out = '';
   188	    let timedOut = false;
   189	    /** @type {NodeJS.Timeout | null} */
   190	    let killTimer = null;
   191	
   192	    const onChunk = (/** @type {Buffer} */ c) => {
   193	      out += c.toString();
   194	    };
   195	    child.stdout?.on('data', onChunk);
   196	    child.stderr?.on('data', onChunk);
   197	
   198	    const timeout = setTimeout(() => {
   199	      timedOut = true;
   200	      if (typeof child.pid === 'number') {
   201	        killGroup(child.pid, 'SIGTERM');
   202	        killTimer = setTimeout(() => {
   203	          if (typeof child.pid === 'number') killGroup(child.pid, 'SIGKILL');
   204	        }, opts.graceMs);
   205	      }
   206	    }, opts.timeoutMs);
   207	
   208	    const finish = (/** @type {number | null} */ exitCode) => {
   209	      clearTimeout(timeout);
   210	      if (killTimer) clearTimeout(killTimer);
   211	      // Belt-and-braces: reap any group survivors even on a clean exit path.
   212	      if (timedOut && typeof child.pid === 'number') killGroup(child.pid, 'SIGKILL');
   213	      try {
   214	        mkdirSync(dirname(opts.logPath), { recursive: true });
   215	        writeFileSync(opts.logPath, capLog(out, maxLogBytes));
   216	      } catch {
   217	        // The log must never mask the result.
   218	      }
   219	      const parsed = parseVerdict(out, { platform: opts.platform ?? 'codex' });
   220	      resolve({
   221	        timedOut,
   222	        exitCode: timedOut ? null : exitCode,
   223	        verdict: parsed.verdict,
   224	        findings: parsed.findings,
   225	        ...(parsed.reason ? { reason: parsed.reason } : {}),
   226	        logPath: opts.logPath,
   227	        pgid,
   228	      });
   229	    };
   230	
   231	    child.on('error', () => finish(null));
   232	    child.on('close', (code) => finish(code));
   233	  });
   234	}
