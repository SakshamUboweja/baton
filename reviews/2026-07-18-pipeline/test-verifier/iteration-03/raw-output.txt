    90	/**
    91	 * Preflight before a child runs in a seat: the worktree is listed, its index
    92	 * is clean, and HEAD sits on the expected branch. Refusals name the failed
    93	 * check and never issue destructive git.
    94	 * @param {string} root @param {{seat: string, branch: string}} target @param {any} io
    95	 * @returns {Promise<{ok: true} | {ok: false, check: string, refusal: string}>}
    96	 */
    97	export async function preflightWorktree(root, target, io) {
    98	  const path = worktreePaths(root).seats[target.seat];
    99	  const listed = parseWorktreeList((await git(io, root, ['worktree', 'list', '--porcelain'])).stdout);
   100	  if (!listed.some((e) => e.path === path)) {
   101	    return { ok: false, check: 'missing-worktree', refusal: `worktree ${path} is not listed by git — run setup or self-heal first` };
   102	  }
   103	  const status = (await git(io, path, ['status', '--porcelain'])).stdout.trim();
   104	  if (status.length > 0) {
   105	    return { ok: false, check: 'dirty-index', refusal: `worktree ${path} has a dirty index/working tree — a clean seat is required before a child runs` };
   106	  }
   107	  const head = (await git(io, path, ['rev-parse', '--abbrev-ref', 'HEAD'])).stdout.trim();
   108	  if (head !== target.branch) {
   109	    return { ok: false, check: 'wrong-branch', refusal: `worktree ${path} HEAD is on '${head}', expected branch '${target.branch}'` };
   110	  }
   111	  return { ok: true };
   112	}

   114	/**
   115	 * Postflight after a child: HEAD still on the expected branch (checked in the
   116	 * seat cwd) and main unmoved from the pre-child capture.
   117	 * @param {string} root @param {{seat: string, branch: string, mainSha: string}} target @param {any} io
   118	 * @returns {Promise<{ok: true} | {ok: false, check: string, refusal: string}>}
   119	 */
   120	export async function postflightWorktree(root, target, io) {
   121	  const path = worktreePaths(root).seats[target.seat];
   122	  const head = (await git(io, path, ['rev-parse', '--abbrev-ref', 'HEAD'])).stdout.trim();
   123	  if (head !== target.branch) {
   124	    return { ok: false, check: 'head-drift', refusal: `the child left HEAD on '${head}' instead of the expected branch '${target.branch}'` };
   125	  }

   127	  if (main !== target.mainSha) {
   128	    return { ok: false, check: 'main-moved', refusal: `main moved under the child (was ${target.mainSha}, now ${main}) — only the merger writes main` };
   129	  }
   130	  return { ok: true };
   131	}

   133	/**
   134	 * Full-range attribution scan over main..branch — the merge gate. Every
   135	 * commit must be authored AND committed by the sole author, with no forbidden
   136	 * AI-attribution trailer in the body. The range is never windowed.
   137	 * @param {string} root @param {string} branch @param {any} io
   138	 * @returns {Promise<{ok: true} | {ok: false, offending: string, reason: string}>}
   139	 */
   140	export async function attributionScan(root, branch, io) {
   141	  const out = (await git(io, root, ['log', `main..${branch}`, `--format=${LOG_FORMAT}`])).stdout;
   142	  for (const record of String(out).split('\x1e')) {
   143	    if (record.trim().length === 0) continue;
   144	    const [sha, an, ae, cn, ce, body = ''] = record.replace(/^\n/, '').split('\x00');
   145	    if (an !== SOLE_AUTHOR.name || ae !== SOLE_AUTHOR.email) {
   146	      return { ok: false, offending: sha, reason: `commit ${sha} has a foreign author identity (${an} <${ae}>) — sole-author is required` };
   147	    }
   148	    if (cn !== SOLE_AUTHOR.name || ce !== SOLE_AUTHOR.email) {
   149	      return { ok: false, offending: sha, reason: `commit ${sha} has a foreign committer identity (${cn} <${ce}>) — sole-author is required` };
   150	    }
   151	    if (TRAILER_PATTERNS.some((re) => re.test(body))) {
   152	      return { ok: false, offending: sha, reason: `commit ${sha} carries a forbidden AI-attribution trailer (co-authored/generated-with/emoji) in its message` };
   153	    }
   154	  }
   155	  return { ok: true };
   156	}

   158	const mergeLockPath = (/** @type {string} */ root) => `${root}/.handoff/loop/merge.lock`;
   159	
   160	/**
   161	 * Merge an approved subtask branch into main — one guarded transaction:
   162	 * acquire the merge lock (a held lock refuses), gate on the attribution scan,
   163	 * merge at the repo root, abort-and-park on conflict, then ff-only sync both
   164	 * seats. The lock is released on EVERY exit path after acquisition.
   165	 * @param {string} root @param {{branch: string}} input @param {any} io
   166	 * @returns {Promise<{ok: true, branch: string} | {ok: false, reason: string}>}
   167	 */
   168	export async function mergeSubtask(root, input, io) {
   169	  const lock = mergeLockPath(root);
   170	  if (io.fs.existsSync(lock)) {
   171	    return { ok: false, reason: `the merge lock at ${lock} is held by another merger — refusing a concurrent merge` };
   172	  }
   173	  ensureDir(io.fs, `${root}/.handoff/loop`);
   174	  atomicWriteJson(io.fs, lock, { host: io.host, pid: io.pid, at: io.now() });
   175	  try {
   176	    const scan = await attributionScan(root, input.branch, io);
   177	    if (scan.ok !== true) return { ok: false, reason: scan.reason };
   178	
   179	    try {
   180	      await git(io, root, ['merge', input.branch]);
   181	    } catch (err) {
   182	      // Conflict (or any merge failure): abort, never auto-resolve — park.
   183	      try {
   184	        await git(io, root, ['merge', '--abort']);
   185	      } catch {
   186	        // Nothing to abort — the failure predates a merge state.
   187	      }
   188	      const msg = /** @type {any} */ (err)?.stdout || /** @type {any} */ (err)?.message || String(err);
   189	      return { ok: false, reason: `merge conflict — aborted and parked for the operator: ${String(msg).split('\n')[0]}` };
   190	    }
   191	
   192	    const seats = worktreePaths(root).seats;
   193	    for (const seat of SEATS) {
   194	      try {
   195	        await git(io, seats[seat], ['merge', '--ff-only', 'main']);
   196	      } catch (err) {
   197	        const msg = /** @type {any} */ (err)?.stderr || /** @type {any} */ (err)?.message || String(err);
   198	        return { ok: false, reason: `worktree wt-${seat} could not fast-forward to main — corruption signal, parked: ${String(msg).split('\n')[0]}` };
   199	      }
   200	    }
   201	    return { ok: true, branch: input.branch };
   202	  } finally {
   203	    try {
   204	      if (io.fs.existsSync(lock)) io.fs.unlinkSync(lock);
   205	    } catch {
   206	      // Best effort — a leaked lock is visible and operator-recoverable.
   207	    }
   208	  }
   209	}
