    94	    return entries.map((e) => `worktree ${e.path}\nHEAD ${MAIN_SHA}\nbranch refs/heads/${e.branch}\n`).join('\n');
    95	  };
    96	
    97	  const fn = (/** @type {string} */ cmd, /** @type {string[]} */ args = [], /** @type {any} */ opts = {}) => {
    98	    const argstr = args.join(' ');
    99	    calls.push({ cmd, args, argstr, cwd: opts?.cwd, lockPresent: lockProbeRef.fn ? lockProbeRef.fn() : undefined });
   100	    if (cmd !== 'git') return Promise.reject(Object.assign(new Error('spawn ENOENT'), { code: 'ENOENT' }));
   101	
   102	    // Explicit overrides win.
   103	    const okey = Object.keys(over).sort((a, b) => b.length - a.length).find((k) => argstr.includes(k));
   104	    if (okey) {
   105	      const r = over[okey];
   106	      if (r?.reject) return Promise.reject(Object.assign(new Error(r.stderr || 'git failed'), { code: r.code ?? 1, stdout: r.stdout ?? '', stderr: r.stderr ?? '' }));
   107	      return Promise.resolve({ stdout: r?.stdout ?? '', stderr: r?.stderr ?? '' });
   108	    }
   109	
   110	    if (argstr.startsWith('worktree list')) return Promise.resolve({ stdout: list(), stderr: '' });
   111	    if (argstr.startsWith('worktree add')) {
   112	      const path = args.find((a) => a.startsWith('/repo/.worktrees/'));
   113	      const bi = args.indexOf('-b');
   114	      const branch = bi >= 0 ? args[bi + 1] : 'baton/wt-x/base';
   115	      if (path) { added.set(path, branch); branchAt.set(path, branch); }
   116	      return Promise.resolve({ stdout: '', stderr: '' });
   117	    }
   118	    if (argstr.startsWith('worktree prune') || argstr.startsWith('worktree remove')) return Promise.resolve({ stdout: '', stderr: '' });
   119	    if (argstr.startsWith('checkout -b')) { if (opts?.cwd) branchAt.set(opts.cwd, args[args.indexOf('-b') + 1]); return Promise.resolve({ stdout: '', stderr: '' }); }
   120	    if (argstr.startsWith('status --porcelain')) return Promise.resolve({ stdout: '', stderr: '' });
   121	    if (argstr === 'rev-parse --abbrev-ref HEAD') return Promise.resolve({ stdout: `${branchAt.get(opts?.cwd) ?? 'main'}\n`, stderr: '' });
   122	    if (argstr.startsWith('rev-parse')) return Promise.resolve({ stdout: `${MAIN_SHA}\n`, stderr: '' });
   123	    if (argstr.startsWith('log ')) return Promise.resolve({ stdout: log, stderr: '' });
   124	    if (argstr.startsWith('merge')) return Promise.resolve({ stdout: 'Fast-forward', stderr: '' });
   125	    if (argstr.startsWith('branch -D')) return Promise.resolve({ stdout: '', stderr: '' });
   126	    return Promise.reject(Object.assign(new Error(`unstubbed git: ${argstr}`), { code: 'ENOSTUB' }));
   127	  };
   128	  fn.calls = calls;
   129	  fn.git = () => calls.filter((c) => c.cmd === 'git');
   130	  fn.issued = (/** @type {RegExp} */ re) => calls.some((c) => c.cmd === 'git' && re.test(c.argstr));
   131	  fn.matching = (/** @type {RegExp} */ re) => calls.filter((c) => c.cmd === 'git' && re.test(c.argstr));
   132	  fn.withCwd = (/** @type {RegExp} */ re, /** @type {string} */ cwd) => calls.filter((c) => c.cmd === 'git' && re.test(c.argstr) && c.cwd === cwd);
   133	  fn.setLockProbe = (/** @type {() => boolean} */ f) => { lockProbeRef.fn = f; };
   134	  return fn;
   135	}

   137	function fakeRunner(io, script) {
   138	  const queue = [...script];
   139	  const calls = /** @type {any[]} */ ([]);
   140	  const fn = (/** @type {any} */ spec, /** @type {any} */ opts = {}) => {
   141	    calls.push({ spec, opts, command: spec?.command, args: spec?.args ?? [], env: spec?.env });
   142	    const r = queue.shift() ?? { verdict: 'APPROVED' };
   143	    if (typeof r.logContent === 'string' && opts.logPath) {
   144	      const dir = opts.logPath.slice(0, opts.logPath.lastIndexOf('/'));
   145	      io.fs.mkdirSync(dir, { recursive: true });
   146	      io.fs.writeFileSync(opts.logPath, r.logContent);
   147	    }
   148	    return Promise.resolve({ timedOut: false, exitCode: r.exitCode ?? 0, verdict: r.verdict ?? 'APPROVED', findings: r.findings ?? '', logPath: opts.logPath });
   149	  };
   150	  fn.calls = calls;
   151	  return fn;
   152	}

    72	/**
    73	 * The pure transition reducer. Never mutates its input; unknown events and
    74	 * refused transitions return the input state unchanged (a new reference is
    75	 * only produced for a real transition).
    76	 * @param {any} state @param {any} ev
    77	 */
    78	export function applyLoopEvent(state, ev) {
    79	  switch (ev?.type) {
    80	    case LOOP_EVENT.PHASE_ADVANCE: {
    81	      const next = state.phaseIndex + 1;
    82	      if (next >= state.phaseCount) return { ...state, phaseIndex: state.phaseCount, status: LOOP_STATUS.DONE };
    83	      return { ...state, phaseIndex: next };
    84	    }
    85	    case LOOP_EVENT.GATE_ITERATION: {
    86	      const gate = String(ev.gate);
    87	      const current = state.iterations?.[gate] ?? 0;
    88	      // The 6th attempt is REFUSED: the counter stays at the cap and the run
    89	      // escalates with a record naming the gate — never a 6th run.
    90	      if (current >= GATE_ITERATION_CAP) {
    91	        return { ...state, status: LOOP_STATUS.ESCALATED, escalation: { gate, iteration: current } };
    92	      }
    93	      return { ...state, iterations: { ...state.iterations, [gate]: current + 1 } };
    94	    }
    95	    case LOOP_EVENT.PARK:
    96	      return { ...state, status: LOOP_STATUS.PARKED, parkReason: ev.reason ?? null };
    97	    case LOOP_EVENT.RESUME:
    98	      // Only a PARK is resumable; an escalation is an operator decision.
    99	      if (state.status !== LOOP_STATUS.PARKED) return state;
   100	      return { ...state, status: LOOP_STATUS.RUNNING, parkReason: null };
   101	    case LOOP_EVENT.SMOKE_AWAIT:
   102	      return { ...state, status: LOOP_STATUS.AWAITING_SMOKE_APPROVAL };
   103	    case LOOP_EVENT.SMOKE_APPROVE: {
   104	      // Guarded transition: only meaningful while awaiting, and only with the
   105	      // caller's verifySmokeToken result stamped verified === true. A missing
   106	      // or false `verified` is unverified — it never resumes the run.
   107	      if (state.status !== LOOP_STATUS.AWAITING_SMOKE_APPROVAL) return state;
   108	      if (ev.verified !== true) return state;
   109	      return { ...state, status: LOOP_STATUS.RUNNING, smokeApproval: { token: ev.token, at: ev.ts ?? null } };
   110	    }
   111	    default:
   112	      return state;
   113	  }
   114	}

   158	/**
   159	 * Load = snapshot + torn-tail-tolerant replay of journal events PAST the
   160	 * snapshot's seq. Duplicate seqs/dedupeKeys apply exactly once, so a
   161	 * supervisor-death recovery never re-runs a completed iteration.
   162	 * @param {string} root @param {any} io
   163	 * @returns {Promise<{state: any, warnings: string[]}>}

   166	  const p = loopPaths(root);
   167	  /** @type {string[]} */
   168	  const warnings = [];

   170	  if (!snap.ok) return { state: null, warnings: ['no loop state snapshot found'] };
   171	  let state = snap.value;
   172	  const baseSeq = typeof state.journalSeq === 'number' ? state.journalSeq : 0;
   173	
   174	  if (io.fs.existsSync(p.journal)) {
   175	    const { entries, warnings: jw } = readAllTolerant(io.fs, p.journal);
   176	    for (const w of jw ?? []) warnings.push(`journal: ${w.kind ?? 'warning'} at line ${w.line ?? '?'}`);
   177	    const applied = new Set();
   178	    let maxSeq = baseSeq;
   179	    for (const e of entries) {
   180	      const seq = typeof e.seq === 'number' ? e.seq : null;
   181	      if (seq !== null && seq <= baseSeq) continue; // already folded into the snapshot
   182	      const key = e.dedupeKey ?? (seq !== null ? `seq:${seq}` : null);
   183	      if (key !== null && applied.has(key)) continue; // replayed duplicate — apply once
   184	      if (seq !== null && applied.has(`seq:${seq}`)) continue;

   186	      if (key !== null) applied.add(key);
   187	      if (seq !== null) {
   188	        applied.add(`seq:${seq}`);
   189	        if (seq > maxSeq) maxSeq = seq;
   190	      }
   191	    }
   192	    state = { ...state, journalSeq: maxSeq };
   193	  }
   194	  return { state, warnings };

   213	/**

   217	 * @param {Record<string, string>} inputs

   220	export function smokeApprovalToken(inputs) {
   221	  for (const f of SMOKE_TOKEN_FIELDS) {
   222	    if (typeof inputs?.[f] !== 'string') throw new Error(`smoke approval token: required input '${f}' is missing`);
   223	  }

   227	/**

   229	 * drifted input. Never throws — partial current inputs simply cannot verify.
   230	 * @param {string} presented @param {Record<string, string>} inputs
   231	 * @returns {{ok: true} | {ok: false, driftedInputs: string[]}}
   232	 */
   233	export function verifySmokeToken(presented, inputs) {
   234	  /** @type {string} */
   235	  let current;
   236	  try {
   237	    current = smokeApprovalToken(inputs);
   238	  } catch {
   239	    return { ok: false, driftedInputs: SMOKE_TOKEN_FIELDS.filter((f) => typeof inputs?.[f] !== 'string') };
   240	  }
   241	  if (presented === current) return { ok: true };
   242	  const a = String(presented ?? '').split('.');
   243	  const b = current.split('.');
   244	  /** @type {string[]} */
   245	  const drifted = [];
   246	  SMOKE_TOKEN_FIELDS.forEach((f, i) => {
   247	    if (a[i + 1] !== b[i + 1]) drifted.push(f);
   248	  });
   249	  return { ok: false, driftedInputs: drifted.length > 0 ? drifted : [...SMOKE_TOKEN_FIELDS] };
   250	}
